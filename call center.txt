call center simulation:

from collections import deque

# Create an empty queue
call_queue = deque()

while True:
    print("\n--- Call Center Queue ---")
    print("1. Add a customer call")
    print("2. Answer and remove the first call")
    print("3. View all waiting calls")
    print("4. Check if queue is empty")
    print("5. Exit")

    choice = input("Enter your choice: ")

    if choice == "1":
        customer = input("Enter customer name: ")
        call_queue.append(customer)
        print(f"Call from {customer} added to the queue.")

    elif choice == "2":
        if len(call_queue) == 0:
            print("No calls waiting to be answered.")
        else:
            customer = call_queue.popleft()
            print(f"Answering call from {customer}.")

    elif choice == "3":
        if len(call_queue) == 0:
            print("No calls are currently waiting.")
        else:
            print("Calls currently waiting:")
            for i, customer in enumerate(call_queue, start=1):
                print(f"{i}. {customer}")

    elif choice == "4":
        if len(call_queue) == 0:
            print("The queue is empty.")
        else:
            print("The queue is not empty.")

    elif choice == "5":
        print("Exiting Call Center Queue.")
        break

    else:
        print("Invalid choice. Please try again.")
